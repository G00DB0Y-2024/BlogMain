> The detailed evaluating procedure of DeepPQL is outlined in Algorithm \ref{alg:deep_pql_cpt}.
> \begin{algorithm}[tb]
> \caption{DeepPQL Evaluation with Consistent Policy Tracking (CPT)}
> \label{alg:deep_pql_cpt}
> \begin{algorithmic}[1]
> \REQUIRE Generative Q-network $Q_\theta$, Initial state $s_0$, Discount factor $\gamma$, Latent sample size $K$
> \STATE \textbf{Initialization phase:}
> \STATE Draw $K$ latent samples $\{\mathbf{z}_k\}_{k=1}^K \sim p(\mathbf{z})$
> \STATE Generate candidate sets and extract the Pareto front for the initial state: 
> $ \mathcal{F}_0 = ND\left(\bigcup_{a \in \mathcal{A}} \{Q_\theta(s_0, a, \mathbf{z}_k)\}_{k=1}^K \right) $
> \STATE Receive the \textit{a posteriori} selected initial target $\mathbf{\hat{q}}_0 \in \mathcal{F}_0$ (e.g., from user preference)
> \STATE Set current step $t = 0$
> \WHILE{episode is not terminated}
>     \STATE Draw $K$ new latent samples $\{\mathbf{z}_k\}_{k=1}^K \sim p(\mathbf{z})$
>     \STATE Select the optimal action $a_t$ by minimizing the Euclidean distance between candidate returns and the current target:
>     $$ a_t = \arg\min_{a \in \mathcal{A}} \left( \min_{k \in \{1, \dots, K\}} \left\| Q_\theta(s_t, a, \mathbf{z}_k) - \mathbf{\hat{q}}_t \right\|_2 \right) $$
>     \STATE Execute action $a_t$ in the environment
>     \STATE Observe immediate reward $\mathbf{r}_t$ and transition to the next state $s_{t+1}$
>     \STATE Calculate the inverse ideal target for the next state via the Bellman algebraic constraint:
>     $\mathbf{\hat{q}}_{t+1} = \frac{\mathbf{\hat{q}}_t - \mathbf{r}_t}{\gamma} $
>     \STATE Update $t \leftarrow t + 1$
> \ENDWHILE
> \end{algorithmic}
> \end{algorithm}
